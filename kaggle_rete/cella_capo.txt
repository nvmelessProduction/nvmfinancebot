CODICE = '/tmp/codice'

if z:
    shutil.unpack_archive(z[0], CODICE)
else:
    d = [os.path.dirname(os.path.dirname(p)) for p in glob.glob('/kaggle/input/**/kaggle_rete/capo.py', recursive=True)]
    assert d, "MANCA IL CODICE NUOVO: carica il nuovo nvm-codice.zip come nuova versione del dataset"
    shutil.copytree(d[0], CODICE)
sys.path.insert(0, CODICE)
